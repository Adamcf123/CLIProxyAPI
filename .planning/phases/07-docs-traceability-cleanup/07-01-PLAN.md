import pathlib
import re
import sys

path = pathlib.Path('.planning/REQUIREMENTS.md')
text = path.read_text(encoding='utf-8')

# Checklist: each METR-01..04 line must be checked.
for i in range(1, 5):
    rid = f"METR-0{i}"
    if not re.search(rf"^.*\[\s*x\s*\].*{re.escape(rid)}.*$", text, flags=re.MULTILINE):
        print(f"missing checked checklist item for {rid}")
        sys.exit(1)

# Traceability table: METR-01..04 must map to Phase 1 with a completed status.
allowed_status = {"complete", "completed", "done", "satisfied"}
rows = []

# Default column indices (matches current REQUIREMENTS.md table).
req_i, phase_i, status_i = 0, 1, 2

for line in text.splitlines():
    if not line.startswith("|"):
        continue
    cols = [c.strip() for c in line.strip().strip("|").split("|")]

    # If a header row is present, learn column positions.
    header = [c.strip().lower() for c in cols]
    if header and header[0] == "requirement" and "phase" in header and "status" in header:
        req_i = header.index("requirement")
        phase_i = header.index("phase")
        status_i = header.index("status")
        continue

    if len(cols) > max(req_i, phase_i, status_i) and cols[req_i].startswith("METR-"):
        rows.append(cols)

by_req = {r[req_i]: r for r in rows}
for i in range(1, 5):
    rid = f"METR-0{i}"
    row = by_req.get(rid)
    if not row:
        print(f"missing traceability row for {rid}")
        sys.exit(1)

    nums = re.findall(r"\d+", row[phase_i])
    if not nums or int(nums[0]) != 1:
        print(f"unexpected phase for {rid}: {row[phase_i]}")
        sys.exit(1)

    # Normalize status like "Complete ✅" -> "complete".
    status = re.sub(r"[^a-z]+", "", row[status_i].lower())
    if status not in allowed_status:
        print(f"unexpected status for {rid}: {row[status_i]}")
        sys.exit(1)

# Last updated marker: require explicit Phase 7 marker.
if "after phase 7 docs cleanup" not in text.lower():
    print("missing last updated marker: 'after phase 7 docs cleanup'")
    sys.exit(1)

print("ok")
