import pathlib
import sys

def must_contain(path: pathlib.Path, needle: str):
    text = path.read_text(encoding='utf-8')
    if needle not in text:
        print(f"missing required text in {path}: {needle}")
        sys.exit(1)

def must_not_contain(path: pathlib.Path, needles_lower):
    text = path.read_text(encoding='utf-8').lower()
    for n in needles_lower:
        if n in text:
            print(f"unexpected '{n}' mention in {path}")
            sys.exit(1)

readme = pathlib.Path('README.md')
readme_cn = pathlib.Path('README_CN.md')

for p in (readme, readme_cn):
    must_contain(p, '## Metrics (TPS/TTFT/TPOT)')
    must_contain(p, 'logs/metrics.db')
    must_contain(p, 'v0/management/metrics')
    # Require at least one of these to avoid hard-coding env var style.
    must_contain(p, 'X-Management-Key')
    must_not_contain(p, ['.jsonl', ' jsonl ', 'jsonl'])

print('ok')
PY
