func TestWriter_DetectsRequestIDConflict(t *testing.T) {
    resetPersistenceHealthForTest()
    
    db := testutil.MustOpenMetricsDB(t)
    defer db.Close()
    
    // Start the writer
    w := newSQLiteWriter(100)
    if err := w.Start(db); err != nil {
        t.Fatalf("failed to start writer: %v", err)
    }
    
    // Create a record with specific request ID
    record := MetricRecord{
        RequestID: "test_conflict_001",
        Provider:  "test",
        Model:     "model",
    }
    
    // First insert should succeed
    w.Enqueue(record)
    time.Sleep(100 * time.Millisecond) // Allow async write
    
    // Verify first insert succeeded (no drop recorded yet)
    health1 := GetPersistenceHealth(time.Now())
    if health1.DroppedTotal != 0 {
        t.Errorf("expected no drops after first insert, got %d", health1.DroppedTotal)
    }
    
    // Second insert with same request_id should trigger conflict
    w.Enqueue(record)
    time.Sleep(100 * time.Millisecond) // Allow async write
    
    // Verify conflict was detected and recorded
    health2 := GetPersistenceHealth(time.Now())
    if health2.DroppedTotal != 1 {
        t.Errorf("expected 1 drop after conflict, got %d", health2.DroppedTotal)
    }
    if health2.LastDropReason == nil {
        t.Fatal("expected LastDropReason to be set after conflict")
    }
    if *health2.LastDropReason != DropReasonRequestIDConflict {
        t.Errorf("expected LastDropReason=%q, got %q", 
            DropReasonRequestIDConflict, *health2.LastDropReason)
    }
    
    // Verify the record exists in DB (first insert succeeded)
    var count int
    err := db.QueryRow("SELECT COUNT(*) FROM metrics WHERE request_id = ?;", 
        record.RequestID).Scan(&count)
    if err != nil {
        t.Fatalf("failed to query DB: %v", err)
    }
    if count != 1 {
        t.Errorf("expected 1 row in DB, got %d", count)
    }
}
```

注意：
1. 使用 resetPersistenceHealthForTest() 清理状态
2. 使用 testutil.MustOpenMetricsDB(t) 创建测试数据库
3. 等待异步写入完成（time.Sleep）
4. 验证冲突被正确记录到 health
5. 验证 DB 中只有一条记录（第一次插入成功，第二次被冲突丢弃）

func TestGetMetrics_ExposesRequestIDConflict(t *testing.T) {
    metricspersist.ResetPersistenceHealthForTest()
    
    db := testutil.MustOpenMetricsDB(t)
    defer db.Close()
    
    // Start the metrics writer
    metricspersist.StartWriter(db)
    
    // Create handler
    h := &Handler{db: db, nowUTC: time.Now}
    
    // Insert a record directly to ensure it exists
    _, err := db.Exec(`
        INSERT INTO metrics (request_id, provider, model, created_at)
        VALUES (?, ?, ?, datetime('now'));`,
        "conflict_test_001", "test", "model")
    if err != nil {
        t.Fatalf("failed to insert test record: %v", err)
    }
    
    // Now enqueue a record with the same request_id to trigger conflict
    metricspersist.Enqueue(metricspersist.MetricRecord{
        RequestID: "conflict_test_001",
        Provider:  "test",
        Model:     "model",
    })
    
    // Wait for async write
    time.Sleep(200 * time.Millisecond)
    
    // Make API request
    gin.SetMode(gin.TestMode)
    w := httptest.NewRecorder()
    c, _ := gin.CreateTestContext(w)
    c.Request = httptest.NewRequest(http.MethodGet, "/v0/management/metrics", nil)
    
    h.GetMetrics(c)
    
    // Parse response
    var resp metricsEnvelope[any]
    if err := json.Unmarshal(w.Body.Bytes(), &resp); err != nil {
        t.Fatalf("failed to unmarshal response: %v", err)
    }
    
    // Verify persistence meta is present and shows conflict
    if resp.Meta.Persistence == nil {
        t.Fatal("expected persistence meta when conflict occurred")
    }
    
    if !resp.Meta.Persistence.Degraded {
        t.Error("expected persistence.degraded=true after conflict")
    }
    
    if resp.Meta.Persistence.DroppedTotal < 1 {
        t.Errorf("expected dropped_total >= 1, got %d", resp.Meta.Persistence.DroppedTotal)
    }
    
    if resp.Meta.Persistence.LastDropReason == nil {
        t.Fatal("expected last_drop_reason to be set")
    }
    
    if *resp.Meta.Persistence.LastDropReason != string(metricspersist.DropReasonRequestIDConflict) {
        t.Errorf("expected last_drop_reason=%q, got %q",
            metricspersist.DropReasonRequestIDConflict, *resp.Meta.Persistence.LastDropReason)
    }
}
```

3. 先直接插入一条记录到 DB，再用相同 request_id enqueue 触发冲突
4. 验证 API 响应包含正确的 persistence meta
